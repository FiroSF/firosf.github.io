### [D번 - Bermuda Rectangle](https://codeforces.com/contest/2257/problem/D)

* 정해랑 다르게 풀었다.
* 다음 둘 중 하나라고 생각했다.
  * 버뮤다 사각형을 스위핑하면서 쿼리들을 갱신하느냐
  * 쿼리를 스위핑하면서 버뮤다 사각형들을 더하느냐
* 둘 다 마땅히 뾰족한 수가 보이지 않았다.
* 그도 그럴 것이, 경우의 수가 4개나 나온다.
  * 쿼리가 포함되거나, 반대거나, 쿼리가 위를 뚫거나, 옆을 뚫거나.
* 그래서 나이브 풀이쪽으로 다시 생각해봤다.
* 이게 제한이 작았다면... 사각형 배치 후 쿼리 안에 있는 사각형칸들을 다 더했을 것이다.
* 제한이 크면 이걸 누적합으로 구할 수 있다.
* 그런데 사각형을 imos로 박자니 누적합의 누적합을 구해야 한다.
* 그러면 그냥 레이지세그로 구간에 싹다 더한 다음, 구간합을 구해주면 된다.
  * 이거 하려면 스위핑하면서 사각형을 적절히 더해야 한다.
* 근데 $S$가 너무 커서 이걸 하려면 좌표압축을 해야 한다.
* 즉, sparse한 레이지 세그가 되는데... 레이지 문제를 거의 안 풀어봐서 레이지 계산하는 코드를 짜는 데 오래 걸렸다.
* 또, 좌표압축까지 같이 있어서 생각보다 구현이 많이 귀찮았다.
* 풀면서 이거 레이지 기본문제 + 좌압이라 많이풀린건가... 생각했다.
* 근데 이게 정해가 아니었다. 정해는 다음과 같다.
  * x축에 따른 사각형 넒이들의 누적합을 구한다.
    * 즉, 가능한 사각형의 x좌표들에 대해 누적합을 구해두면 된다.
  * 그러고 나서 쿼리를 계산한다.
  * 쿼리 사각형은 다음 두 부분으로 나뉜다:
    * 버뮤다 사각형으로 가득 찬 부분
    * 버뮤다 사각형이 쿼리보다 낮은 부분
  * 앞부분은 단순 사각형 넓이계산으로 구한다.
  * 뒷부분은 누적합에서 이분탐색해서 잘 구해준다.

// 2026-08-18 19:28:13
#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 998244353
#define ll long long
#define pii pair<ll, ll>
#define piii pair<ll, pii>

// 구간합
// 좌표압축을 한다
// 좌표압축 각각에 대한 길이 l을 정의한다
// 그러면 세그 노드 각각에 길이 l이 저장된다
// 레이지에 업데이트가 k로 들어온다
// 노드별로 lk만큼 더해준다
// lazy가 k
// 노드값(arr)이 합
// sparse한 레이지합세그인데?
struct Sum {
    static constexpr ll def = 0, l_def = 0;
    static ll merge(ll a, ll b) { return a + b; }
    static void apply(ll &v, ll l, ll len) { v += l * len; }
    static void comp(ll &t, ll l) { t += l; }
};

vector<ll> pos;
template <typename Op> struct LazySeg {
    int siz;
    vector<ll> arr, lazy;

    LazySeg(int n) {
        siz = 1 << (32 - __builtin_clz(n));
        arr.assign(siz << 1, Op::def);
        lazy.assign(siz << 1, Op::l_def);
    }

    void push(int n, int s, int e) {
        if (lazy[n] == Op::l_def)
            return;
        ll len = pos[e] - (s == 0 ? 0 : pos[s - 1]);
        Op::apply(arr[n], lazy[n], len);
        if (s != e) {
            Op::comp(lazy[n * 2], lazy[n]);
            Op::comp(lazy[n * 2 + 1], lazy[n]);
        }
        lazy[n] = Op::l_def;
    }

    // [l, r]
    void update(int l, int r, ll v, int n, int s, int e) {
        push(n, s, e);
        if (r < s || e < l)
            return;
        if (l <= s && e <= r) {
            Op::comp(lazy[n], v);
            push(n, s, e);
            return;
        }
        int m = (s + e) / 2;
        update(l, r, v, n * 2, s, m);
        update(l, r, v, n * 2 + 1, m + 1, e);
        arr[n] = Op::merge(arr[n * 2], arr[n * 2 + 1]);
    }
    void update(int l, int r, ll v) { update(l, r, v, 1, 0, siz - 1); }

    // [l, r]
    ll query(int l, int r, int n, int s, int e) {
        push(n, s, e);
        if (r < s || e < l)
            return Op::def;
        if (l <= s && e <= r)
            return arr[n];
        int m = (s + e) / 2;
        return Op::merge(query(l, r, n * 2, s, m),
                         query(l, r, n * 2 + 1, m + 1, e));
    }
    ll query(int l, int r) { return query(l, r, 1, 0, siz - 1); }
};

int run() {
    ll A, Q;
    cin >> A >> Q;

    vector<ll> v;
    for (int i = 1; i <= (ll)sqrt(A); i++) {
        if (A % i == 0) {
            v.push_back(i);
            v.push_back(A / i);
        }
    }
    pos = v;
    sort(all(v));
    v.erase(unique(all(v)), v.end());
    vector<pair<pii, int>> qv(Q);
    vector<ll> resv(Q);
    for (int i = 0; i < Q; i++) {
        ll x, y;
        cin >> x >> y;
        pos.push_back(y);
        qv[i] = {{x, y}, i};
    }
    sort(all(qv));
    // pos.push_back(0);
    sort(all(pos));
    pos.erase(unique(all(pos)), pos.end());

    LazySeg<Sum> seg(pos.size());
    // x 키우는 스위핑
    // 다음 버뮤다의 x를 들고있음
    // 쿼리점이 들어올때마다 다음 버뮤다를 이 x로 y구해서 레이지에 더하기
    ll bidx = 0, lastx = 0;
    for (auto &&i : qv) {
        ll qx = i.F.F, qy = i.F.S, qidx = i.S;

        // 버뮤다사각형 적용
        while (bidx < (int)v.size() && v[bidx] <= qx) {
            int yidx = lower_bound(all(pos), A / v[bidx]) - pos.begin();
            // cout << yidx << " " << v[bidx] - lastx << endl;
            seg.update(0, yidx, v[bidx] - lastx);
            lastx = v[bidx];
            bidx++;
        }

        // 남은거 적용
        if (bidx != (int)v.size()) {
            int yidx = lower_bound(all(pos), A / v[bidx]) - pos.begin();
            seg.update(0, yidx, qx - lastx);
            lastx = qx;
        }

        // 쿼리
        int yidx = lower_bound(all(pos), qy) - pos.begin();
        resv[qidx] = seg.query(0, yidx);
        // cout << yidx << endl;
        // for (int i = 0; i < (int)pos.size(); i++) {
        //     cout << pos[i] << "," << seg.query(i, i) << " ";
        // }
        // cout << endl;
    }

    // cout << endl;
    for (auto &&i : resv) {
        cout << i << "\n";
        // cout << endl;
    }

    return 0;

### [E번 - Busy Beaver](https://codeforces.com/contest/2257/problem/E)

* 어떤 건물 하나에 대해서 보자.
* 이 건물을 하나씩 지어 볼 거다.
* 이때 $k$층까지 지었을 때의 수익을 $e$, 투자금을 $c$라고 하자.
* $e \le c$를 만족하는, 가장 작은 $k$를 찾는다.
* 여기까지 짓는 데 필요하는 최소금액 $m$을 구한다.
* 이러면, 여기까지 짓는 거는 적어도 $m$원이 있다면 손해일 수가 없다.
  * 수중에 있는 돈이 늘어날 수밖에 없는 상황이기 때문이다.
* 이런 것을 찾으면 $e$와 $c$를 초기화하고, 다시 건물을 올리면서 위를 반복한다.
* 이렇게 찾은 $(k, m)$쌍들은 서로 우선순위 관계가 있다.
  * 아래층을 지어야 위층을 지을 수 있기 때문이다.
* 이 '체인'들을 각 건물에 대해 구하고, 체인의 앞부분을 우선순위 큐에 넣는다.
  * 이때 $m$이 작은 게 먼저 뽑히게 한다.
* 우선순위 큐에서 얘네를 뽑으면서 수중의 돈을 최대한 늘린다.
* 이걸 다 하면 가질 수 있는 최대 돈 $b$가 계산된다.
* 이제 예산 $b$로 각 건물들을 완성시키는 걸 시도하고, 그중 최고 높이를 찾아 출력하면 된다.
* 이득일 수밖에 없는 것을 찾은 다음 최대한 많이 취하면 되는 문제였다.
* **분류**: 정렬, 그리디, 우선순위 큐

#### 고찰

* 아이디어는 거의 10분 이내로 떠올랐는데 구현이 오래걸렸다...
* 업솔빙할때 보니까 말도 안 되는 실수를 많이 했었다.
  * gemini한테 물어보니까 구현실수가 4군데나 있었다...

// 2026-08-18 20:29:06
#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 998244353
#define ll long long
#define pii pair<ll, ll>
#define piii pair<ll, pii>

ll run() {
    ll N, x;
    cin >> N >> x;

    vector<ll> hv(N);
    // {bene, need}
    vector<vector<pii>> chain(N);
    // 해당 체인의 인덱스
    vector<vector<ll>> chainidc(N);
    // {need, idx}
    priority_queue<pii> pq;

    vector<vector<ll>> costvv(N), earnvv(N);
    for (ll now = 0; now < N; now++) {
        ll cnt;
        cin >> cnt;
        hv[now] = cnt;

        vector<ll> costv(cnt), earnv(cnt);
        for (ll i = 0; i < cnt; i++) {
            cin >> costv[i];
        }
        for (ll i = 0; i < cnt; i++) {
            cin >> earnv[i];
        }
        costvv[now] = costv;
        earnvv[now] = earnv;

        ll bene = 0, need = 0;
        for (ll i = 0; i < cnt; i++) {
            need = min(need, bene - costv[i]);
            bene += earnv[i] - costv[i];

            if (bene > 0) {
                chain[now].push_back({bene, need});
                chainidc[now].push_back(i);
                bene = 0;
                need = 0;
            }
        }

        if (chain[now].size())
            pq.push({chain[now][0].S, now});
    }

    ll maxh = -1;
    ll building = 0;
    vector<ll> idc(N);
    while (pq.size()) {
        // need가 안되면 ㅈㅈ
        if (-pq.top().F > x)
            break;

        ll idx = pq.top().S;
        ll cidx = idc[idx];
        pq.pop();
        if (cidx >= (ll)chainidc[idx].size()) {
            continue;
        }

        x += chain[idx][cidx].F;
        idc[idx]++;
        if (maxh < chainidc[idx][cidx] ||
            (maxh == chainidc[idx][cidx] && building > idx)) {
            maxh = max(maxh, chainidc[idx][cidx]);
            building = idx;
        }

        if (idc[idx] < (ll)chain[idx].size()) {
            pq.push({chain[idx][cidx + 1].S, idx});
        }
    }

    for (ll now = 0; now < N; now++) {
        ll cidx = idc[now];
        ll start = 0;
        if (cidx)
            start = chainidc[now][cidx - 1] + 1;
        ll budg = x;
        for (ll i = start; i < hv[now]; i++) {
            if (costvv[now][i] > budg) {
                break;
            }
            if (maxh < i || (maxh == i && building > now)) {
                maxh = max(maxh, i);
                building = now;
            }

            budg = budg + earnvv[now][i] - costvv[now][i];
        }
    }

    cout << maxh + 1 << " " << building + 1 << "\n";
    // cout << flush;
    return 0;
}

signed main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    ll T;
    cin >> T;

    for (ll i = 0; i < T; i++) {
        run();
    }

    return 0;
}
