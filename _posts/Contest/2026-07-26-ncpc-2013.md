### B - Boiling Vegetables (백준 9238 당근 볶기, 정올 15842 채소 삶기)
* 가장 큰 것과 가장 작은 것이 최대한 비슷한 크기여야 한다.
* 따라서 균등한 크기로 자르는 것이 항상 이득이다.
* 그렇다면 투포인터를 해줄 수 있다.
  * 먼저 모든 야채를 $1$조각, $2$조각, $\dots$, $500$조각으로 나누었을 때, $\{\text{조각의 크기}, \text{조각 개수}, \text{야채 번호}\}$를 저장해주자.
  * 이걸 정렬한다.
  * 포인터 $l, r$을 잡는다.
  * $l$을 늘려가면서, $l$에 위치한 조각이 가장 작은 것이라 했을 때 가능한 모든 조각들을 $r$을 옮겨가며 찾아준다.
    * 이 과정에서 중복되는 야채가 나온다면, 조각 수가 적은걸로 해주면 된다.
  * 이러면 $l$이 가장 작은 조각일 때 최적의 선택을 구할 수 있다.
  * 이걸 끝까지 해 주면 된다.
* $500$만 개 정도니까 시간 내로 동작한다.
* 정해는 우선순위 큐로 그리디하게 큰 것부터 조각수를 늘리는 방식인 것 같다.
* TMI: 내 코드를 제미나이한테 주면 할루시네이션이 매우 잘 일어난다.
* **분류**: 정렬, 투포인터

```cpp
#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define ll long long
#define pii pair<int, int>
#define piii pair<int, pii>
#define SIZ 500

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    int N;
    double t;
    cin >> t >> N;

    vector<int> vis(N);
    // {조각크기, {인덱스, 자르는횟수}}
    vector<pair<double, pii>> v;
    for (int i = 0; i < N; i++) {
        double siz;
        cin >> siz;
        for (int j = 1; j <= SIZ; j++) {
            v.push_back({siz / j, {i, j}});
        }
    }

    int hi = 0, cnt = 0, totcnt = 0, res = INF;
    sort(all(v));
    for (int i = 0; i < N * SIZ; i++) {
        double siz = v[i].F;
        double mx = siz / t;
        while (hi < (N * SIZ) && v[hi].F < mx) {
            if (vis[v[hi].S.F] == 0)
                cnt++;
            vis[v[hi].S.F]++;

            if (v[hi].S.S == SIZ) {
                totcnt += SIZ;
            } else {
                totcnt--;
            }
            hi++;
        }

        if (cnt == N) {
            res = min(res, totcnt);
        }

        if (vis[v[i].S.F] == 1) {
            cnt--;
        }
        vis[v[i].S.F]--;
    }

    cout << res - N << "\n";
    return 0;
}
```

```cpp
// 2026-07-23 19:17:10
#include <bits/stdc++.h>
#include <iomanip>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define ll long long
#define pii pair<int, int>
#define piii pair<int, pii>

#define x first
#define y second
using Point = pair<ll, ll>;
Point operator+(Point p1, Point p2) { return {p1.x + p2.x, p1.y + p2.y}; }
Point operator-(Point p1, Point p2) { return {p1.x - p2.x, p1.y - p2.y}; }
ll operator*(Point p1, Point p2) { return p1.x * p2.x + p1.y * p2.y; } // 내적
ll operator/(Point p1, Point p2) { return p1.x * p2.y - p2.x * p1.y; } // 외적
int Sign(ll v) {
    return (v > 0) - (v < 0);
} // 양수면 +1, 음수면 -1, 0이면 0 반환
ll Dist(Point p1, Point p2) { return (p2 - p1) * (p2 - p1); } // 두 점 거리 제곱
ll SignedArea(Point p1, Point p2, Point p3) { return (p2 - p1) / (p3 - p1); }
int CCW(Point p1, Point p2, Point p3) { return Sign(SignedArea(p1, p2, p3)); }

// 모든 점을 포함하는 가장 작은 볼록 다각형, O(N log N)
vector<Point> ConvexHull(vector<Point> points) {
    if (points.size() <= 1)
        return points;
    swap(points[0], *min_element(points.begin(), points.end()));
    sort(points.begin() + 1, points.end(), [&](auto a, auto b) {
        int dir = CCW(points[0], a, b);
        if (dir != 0)
            return dir > 0;
        return Dist(points[0], a) < Dist(points[0], b);
    });
    vector<Point> hull;
    for (auto p : points) {
        while (hull.size() >= 2 &&
               CCW(hull[hull.size() - 2], hull.back(), p) <= 0)
            hull.pop_back();
        hull.push_back(p);
    }
    return hull;
}

// 가장 먼 두 점을 구하는 함수, O(N)
// 주의: hull은 반시계 방향으로 정렬된 볼록 다각형이어야 함
pair<Point, Point> Calipers(vector<Point> hull) {
    int n = hull.size();
    ll mx = 0;
    Point a, b;
    for (int i = 0, j = 0; i < n; i++) {
        while (j + 1 < n &&
               (hull[i + 1] - hull[i]) / (hull[j + 1] - hull[j]) >= 0) {
            ll now = Dist(hull[i], hull[j]);
            if (now > mx)
                mx = now, a = hull[i], b = hull[j];
            j++;
        }
        ll now = Dist(hull[i], hull[j]);
        if (now > mx)
            mx = now, a = hull[i], b = hull[j];
    }
    return {a, b};
}

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    int N;
    cin >> N;

    vector<Point> dots(N);
    for (int i = 0; i < N; i++) {
        cin >> dots[i].F >> dots[i].S;
    }

    auto hull = ConvexHull(dots);
    auto points = Calipers(hull);
    cout << setprecision(10) << fixed << sqrt(Dist(points.F, points.S));

    return 0;
}
```

### E - Virus Replication (백준 9241 바이러스 복제, 정올 15845 바이러스 복제)
* 지문이 비문학이다.
* 대충 일부 구간을 원하는 문자열로 바꿀 수 있을 때, 주어진 문자열을 만들기 위해서 필요한 '원하는 문자열'의 최소 길이를 구하라는 문제이다.
* 뇌절해서 매개변수탐색 + 라빈카프를 짜버렸다...
* 사실은 그냥 왼쪽부터 일치하는 최대길이만큼 가고, 오른쪽도 마찬가지로 해 준 다음, 사이에 있는것만 바꾸면 된다.

```cpp
#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define MOD2 998244353
#define ll long long
#define pii pair<int, int>
#define piii pair<int, pii>

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    string s1, s2;
    cin >> s1 >> s2;

    int N = s1.size(), M = s2.size();

    ll tmp = 0;
    vector<ll> v1l(N + 1), v2l(M + 1);
    for (int i = 0; i < N; i++) {
        tmp = tmp * MOD2 + (s1[i] - 'A');
        tmp %= MOD;
        v1l[i + 1] = tmp;
    }

    tmp = 0;
    for (int i = 0; i < M; i++) {
        tmp = tmp * MOD2 + (s2[i] - 'A');
        tmp %= MOD;
        v2l[i + 1] = tmp;
    }

    tmp = 0;
    vector<ll> v1r(N + 1), v2r(M + 1);
    for (int i = N - 1; i >= 0; i--) {
        tmp = tmp * MOD2 + (s1[i] - 'A');
        tmp %= MOD;
        v1r[i] = tmp;
    }
    tmp = 0;
    for (int i = M - 1; i >= 0; i--) {
        tmp = tmp * MOD2 + (s2[i] - 'A');
        tmp %= MOD;
        v2r[i] = tmp;
    }

    // 사이 길이로 매탐
    ll lo = max(-1, N - M - 1), hi = N + 1;
    while (lo + 1 < hi) {
        // s1 삭제
        ll mid = (lo + hi) / 2;
        // s2 삭제
        ll mid2 = mid - (N - M);
        bool flag = false;
        for (int i = 0; i <= N - mid; i++) {
            if (v1l[i] == v2l[i] && v1r[i + mid] == v2r[i + mid2])
                flag = true;
        }

        if (flag) {
            hi = mid;
        } else {
            lo = mid;
        }
    }

    cout << hi - (N - M);
    return 0;
}
```

### I - Dance Reconstruction (백준 9245 댄스, 정올 15849 춤 복원)
* 순열을 함수로 보자. 이걸 $f$라고 하자.
* 문제는 $f^k$ ($f$를 $k$번 합성)에다가 $1, 2, 3, \dots, N$을 넣었을 때 나오는 수열을 가지고 $f$를 복원하라는 것이다.
* 관찰을 아주 잘 해 보면, $f$에서 크기 $2x$짜리 사이클은 $k=2$일 때 $f^k$에서 크기 $x$짜리 사이클 $2$개로 변한다.
* 일반화하면 대충 크기 $x$짜리 사이클은 $\frac{x}{\gcd(x, k)}$짜리 사이클 $\gcd(x, k)$개로 변한다고 추측 가능하다.
  * 대충 이유를 적어보자면...
  * 역으로 $f^k$에 있는 크기 $\frac{x}{\gcd(x, k)}$짜리 사이클 $\gcd(x, k)$개를 $f$에서 교차하면서 이어붙여주면, 크기 $x$짜리 사이클 하나가 만들어진다.
  * 여기서 $k$만큼 움직이면 사이클 하나에서 $\frac{k}{\gcd(x, k)}$만큼 움직인 것과 같은 효과를 낸다.
  * 따라서, 사이클 각각이 독립적으로 움직이므로 해 구성이 가능하다.
* 그러면... $f^k$를 적용한 수열에서 사이클들을 골라낸 다음, 사이클 길이별로 사이클들을 저장해주자.
* 그 다음, 사이클의 개수에 따라서 $\gcd(x, k)$를 계산해줘서 잘 이어붙이면 될 것이다.
* 그래서 이걸 구현하고 제출했는데, 정올에서 맞았다고 떴다.
* 근데 Kattis에 다시 제출해보니 틀렸다.
* 알고보니 정올 스페셜저지가 뭔가 문제가 있어서 정답처리된 것 같다.
  * 실제로는 $\gcd(x, k)$로 하면 안 된다.
  * 위 예시를 이어서 생각해보자.
    * 사이클 각각이 독립적으로 움직이므로 해 구성이 가능하다고 생각할 수 있는데, 이는 $\frac{k}{\gcd(x, k)}$가 $x$와 서로소일 때만 가능하다.
      * 서로소가 아니면 여기서 또 작은사이클로 나뉘게 된다.
    * 구체적인 예시로, $N=8$, $K=4$, 수열은 $5, 6, 7, 8, 1, 2, 3, 4$가 있다.
    * 여기서는 크기 $2$짜리 사이클이 $4$개니까, $\frac{k}{\gcd(x, k)} = \frac{4}{\gcd(2, 4)} = 2$가 나온다.
    * $\gcd(2, 4) = 2$니까 사이클 $2$개를 이어준다는 얘기인데, 여기서 $4$씩 가버리면 크기 $4$에서 $4$칸씩 가기 때문에 크기 $1$짜리 사이클 네 개로 나뉜다.
    * 따라서, 여기서는 $\gcd(2, 4) = 2$ 대신에 $4$를 써야 한다.
    * 즉, 이 예시에서는 사이클 $4$개를 이어주면 크기 $8$짜리 사이클에서 $4$칸 가는 건 작은 사이클 하나에서 $1$칸 가는 것과 동일하고, $1$과 $2$는 서로소라 만족하는 것이다.
    * 따라서, $K$를 나눈 값이 사이클 개수와 서로소가 되도록 하는 값을 잘 찾아서 그것만큼 사이클들을 이어주면 된다.
* 마지막으로 $f$를 복원할 때 모듈로 역원이 필요하다.
  * $K$만큼 가면 작은 사이클에서 $1$칸 간 게 아니라 여러 칸 간 것이 나올 수 있어서, $nK \equiv 1 \pmod{l}$인 $n$을 찾아줘야 다음 수를 찾을 수 있다.

  * 나는 여기서 오버플로우나서 $1$틀 적립했었다...

```cpp
        v[i]--;

    map<int, vector<vector<int>>> mp;
    vector<bool> vis(N);
    vector<int> resv(N);
    for (int i = 0; i < N; i++) {
        if (vis[i])
            continue;
        vector<int> cycle;
        int now = i;
        while (!vis[now]) {
            cycle.push_back(now);
            vis[now] = true;
            now = v[now];
        }

        if (cycle.size() == 1) {
            resv[i] = i;
            continue;

        ll gcdv = 1;

        while (gcd(l, tmpk) != 1) {
            gcdv *= gcd(l, tmpk);
            tmpk /= gcd(l, tmpk);
        }

        if (cnt % gcdv) {
            cout << "Impossible";
            return 0;

        // gcd 개만큼 나눠주기
        for (int i = 0; i < cnt / gcdv; i++) {

            // 이걸 k라하면 nk = 1 mod l인 n 찾으면댐
            ll k = K / gcdv;
            ll n = 1;
            while (n * k % l != 1) {
                n++;

            vector<vector<int>> cycles;
            for (int j = i * gcdv; j < (i + 1) * gcdv; j++) {

                vector<int> cycle;

                    cycle.push_back(t.S[j][k * n % l]);
                }
                cycles.push_back(cycle);

            vector<int> cycle;
            for (int k = 0; k < l; k++) {
                for (int j = 0; j < gcdv; j++) {
                    cycle.push_back(cycles[j][k]);
                }

                resv[cycle[j]] = cycle[(j + 1) % siz];
                st.insert(cycle[j]);
            }

            totcnt += siz;
        }
    }

    for (auto t : resv) {
        cout << t + 1 << " ";
    }

    return 0;
}
```

### J - Dartboard (백준 9246 다트판, 정올 15850 다트판)
* 확률밀도함수로 적분하라는 문제다.
* 난 정규분포 비슷하게 생긴 저걸 직접 적분식을 구할 생각이 없다.
* 그래서 리만적분 비슷하게? 해줬다. 미적분시간때 배웠는데 이름이 뭐드라
  * 도넛 모양?을 도넛 모양 $100$만 개로 나눠준다.
  * 그러면 각 도넛 모양은 두께가 엄청 얇을 것이다.
  * 도넛 하나의 두께를 $t$라고 하면, 도넛의 넓이는 대충 원의 둘레길이 $\times t$ 정도 된다.
  * 여기에 확률밀도함수를 곱하면 여기 맞을 확률이 된다.
  * 이거 $100$만 개를 다 더해주고, 여기에 다트판 점수를 곱해준다.
* 이거를 모든 구간에 대해 해 주고, 다 더하면 된다.
* **분류**: 확률, 미적분

```cpp
// 2026-07-24 22:07:54
#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define ll long long
#define ld long double
#define pii pair<int, int>
#define piii pair<int, pii>
#define PI 3.14159265358979323846L
#define SIZ 1000000

ld sigma;
ld f(ld r) {
    // f(r) = 1/(2πσ^2) e^(-r^2/(2σ^2))
    return 1 / (2 * PI * sigma * sigma) * expl(-r * r / (2 * sigma * sigma));
}

ld ff(ld r) { return 2 * PI * r * f(r); }

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    vector<ld> rv(7);
    for (int i = 0; i < 6; i++) {
        cin >> rv[i + 1];
    }
    cin >> sigma;

    vector<ld> wei = {50, 25, 10.5, 31.5, 10.5, 21};
    ld res = 0;
    for (int i = 0; i < 6; i++) {
        ld l = rv[i], r = rv[i + 1];
        ld half = (r - l) / (SIZ * 2);
        ld sumv = 0;
        for (int step = 0; step < SIZ; step++) {
            ld p = l + half * (step * 2 + 1);
            sumv += ff(p) * (half * 2);
        }
        res += sumv * wei[i];
    }

    cout << setprecision(10) << fixed << res;
    return 0;
}
```

### K - Cliff Walk (백준 9247 절벽 걷기, 정올 15851 절벽 산책)
* 지문이 비문학이다.
* 마르는 시간이 $1$시간이라서, 접근 가능한 시간 구간의 중심이 항상 $6.5$다.
* 이걸 활용해서, 시작 지점에서 어느 위치로 갈 때까지 걸리는 최소시간 다익을 돌려줄 수 있다.
* 반대로, 어떤 지점에서 최소 몇 시에 출발해야 도착지에 도달할 수 있는지 찾는 것도 다익으로 찾을 수 있다.
* 이렇게 찾은 두 시간배열을 비교해서, 가는데 걸리는 시간이 최소 출발시각보다 작거나 같으면 도달 가능하다.
* 이런 칸들 중에서 시작 지점부터의 거리 최대를 구하면 된다.
* 부동소수점 이슈 때문에 정말 고생을 많이 했다. 왜 문제를 이렇게 낸 걸까...
* 심지어 정올에서 아무리해도 틀렸다고 나와서 kattis에 제출했더니 맞았다고 떴다.
* 문제 아이디어 자체는 좋은데 부동소수점 이슈 때문에 이번 셋에서 가장 맘에 안 들었던 문제
* **분류**: 구현, 다익스트라

```cpp
// 2026-07-24 23:12:21
#include <bits/stdc++.h>
#include <cassert>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define ll long long
#define ld long double
#define pii pair<int, int>
#define piii pair<int, pii>
#define pdd pair<ld, ld>
#define PI 3.14159265358979323846L
#define e 0.000000001

vector<pii> DIR = {{0, 1}, {-1, 0}, {0, -1}, {1, 0}};
ld a, m;
int N, M, sr, sc;
ld f(ld t) { return 0.5 * a * (cos(t * PI / 6) + 1); }

ld bs(ld lo, ld hi, ld h) {
    for (int i = 0; i < 50; i++) {
        ld mid = (lo + hi) / 2;
        if (f(mid) < h) {
            lo = mid;
        } else {
            hi = mid;
        }
    }

    return (lo + hi) / 2;
}

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);

    cin >> a >> m;
    m /= 3600;

    cin >> M >> N >> sc >> sr;

    vector<vector<ld>> hmap(N, vector<ld>(M));
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            cin >> hmap[i][j];
            hmap[i][j] /= 1000;
        }
    }

    vector dry(N, vector<pdd>(M));
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            if (hmap[i][j] > a) {
                dry[i][j] = {-INF, INF};
                continue;
            }

            // 썰물 밀물
            ld t1 = bs(6, 0, hmap[i][j]);
            ld t2 = bs(6, 12, hmap[i][j]);
            dry[i][j] = {t1 + 1, t2};
        }
    }

    int cnt = 0;
    // 가는데 걸린시간
    vector vis1(N, vector<ld>(M, INF));
    // time, pos
    priority_queue<pair<ld, pii>> pq;
    pq.push({0, {sr, sc}});
    vis1[sr][sc] = 0;
    while (pq.size()) {
        cnt++;
        assert(cnt < N * M * 20);
        auto now = pq.top();
        pq.pop();

        int r = now.S.F, c = now.S.S;
        ld cost = -now.F;
        if (cost > vis1[r][c] + e)
            continue;

        for (auto dir : DIR) {
            int newr = r + dir.F, newc = c + dir.S;
            if (0 > newr || newr >= N || 0 > newc || newc >= M ||
                abs(hmap[newr][newc] - hmap[r][c]) - e > 1)
                continue;
            ld newcost = max(cost, dry[newr][newc].F) + m;

            if (vis1[newr][newc] < newcost + e)
                continue;
            vis1[newr][newc] = newcost;
            pq.push({-newcost, {newr, newc}});
        }
    }

    // 오려면 최소 이시간보다 일찍출발
    vector vis2(N, vector<ld>(M, -INF));
    pq.push({12, {sr, sc}});
    vis2[sr][sc] = 12;
    while (pq.size()) {
        auto now = pq.top();
        pq.pop();

        int r = now.S.F, c = now.S.S;
        ld cost = now.F;
        if (cost + e < vis2[r][c])
            continue;

        for (auto dir : DIR) {
            int newr = r + dir.F, newc = c + dir.S;
            if (0 > newr || newr >= N || 0 > newc || newc >= M ||
                abs(hmap[newr][newc] - hmap[r][c]) - e > 1)
                continue;
            ld newcost = min(cost, dry[newr][newc].S) - m;

            if (vis2[newr][newc] + e > newcost)
                continue;
            vis2[newr][newc] = newcost;
            pq.push({newcost, {newr, newc}});
        }
    }

    ld res = 0;
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < M; j++) {
            if (vis1[i][j] + e <= vis2[i][j]) {
                int dr = abs(i - sr), dc = abs(j - sc);
                res = max(res, sqrtl(dr * dr + dc * dc) * 10);
            }
        }
    }

    cout << setprecision(10) << fixed << res;
    return 0;
}
```
