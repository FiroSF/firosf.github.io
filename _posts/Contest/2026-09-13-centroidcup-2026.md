## 팀 구성

센트컵 시작 2주 전, dongggle님 쪽에서 먼저 팀 제의 연락이 왔다. 문제는 내가 군인 신분이라 일정을 맞추기 상당히 빠듯했다는 점. 그래서 급하게 일정을 수정해서 어떻게든 대회 당일에 참가할 수 있게 되었다.

팀원은 다음과 같다.

* <a href="https://codeforces.com/profile/dongggle" style="color: #ff9900; font-weight: bold;">dongggle</a> - 2023 IUPC 2위.
* <a href="https://codeforces.com/profile/lapinism" style="color: #00aaff; font-weight: bold;">lapinism</a> - 2025 IUPC 1위이자 현 CTP 회장.
* <a href="https://codeforces.com/profile/Firo_SF" style="color: #aa00ff; font-weight: bold;">Firo_SF</a> - 2024 IUPC 2위, 본인이다.

팀을 만들고 가장 먼저 한 것은 팀명 정하기였다. 팀명 흥신소68은 dongggle님의 아이디어이다. 나야 뭐 블로그 이름도 블루아카에서 따왔다보니 만족했는데 lapinism님은 어떻게 생각했는지 모르겠다.

### [D번 🎲 Alea jacta est!](https://doj.kr/ko/problems/508)

* 어떤 면이 나와도 or한 값이 $S$여야 한다.
* 주사위가 $K$개 있다고 하자.
* 주사위 하나에서 어떤 면이 나와도 상관없으려면, 주사위 하나가 비트를 확정시킬 수 있어야 한다.
  * 주사위 하나에 있는 모든 면을 and하면 어떤 비트를 확정시키는지가 나온다.
* 따라서 $N$개의 서로 다른 수들로 비트를 최대한 많이 확정시킬 수 있어야 한다.
  * 이거는 최대한 적은 수의 비트를 비워서 수 $N$개를 구성하면 된다.
  * 비트 $k$개라면 $2^k$가지 수를 만들 수 있다.
* 따라서 $N \le 2^k$인 $k$를 찾아, 각 주사위당 그만큼만 비트에 $0$을 섞어주면 된다.
* $S$의 켜진 비트 수가 $t$라 하면 $t-k$개 비트가 확정되므로, $\displaystyle \left\lceil \frac{t}{t-k} \right\rceil$개의 주사위가 필요하다.

#include <bits/stdc++.h>

using namespace std;

#define all(a) a.begin(), a.end()
#define F first
#define S second
#define INF 0x3f3f3f3f
#define MOD 1000000007
#define ll long long
#define pii pair<int, int>
#define piii pair<int, pii>

int main() {
    // Fast IO
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);
    cout.tie(NULL);
    int N;
    cin >> N;

    string s;
    cin >> s;

    // dp[i][j][k] = i에서 "CUP"[j]가 k번째 컵 만들떄 {최대, 역추적좌표셋}
    vector dp(N, vector(3, vector(100, vector<int>(4, -INF))));

    // dp[0][0][0] = {0, -1, -1, -1};
    if (s[0] == 'C') {
        dp[0][0][0] = {1, -1, -1, -1};
    }

    for (int i = 1; i < N; i++) {
        for (int j = 0; j < 3; j++) {
            for (int k = 0; k < 100; k++) {
                // 기본적으로 1늘어남
                dp[i][j][k] = {dp[i - 1][j][k][0] + 1, i - 1, j, k};

                if (j == 0) {
                    // C
                    if (k == 0) {
                        if (s[i] == 'C' && dp[i][j][k][0] < 1)
                            dp[i][j][k] = {1, -1, -1, -1};
                    } else {
                        if (s[i] == 'C' &&
                            dp[i][j][k][0] < dp[i - 1][2][k - 1][0] + 1) {
                            dp[i][j][k] = {dp[i - 1][2][k - 1][0] + 1, i - 1, 2,
                                           k - 1};
                        }
                    }
                } else if (j == 1) {
                    // U
                    if (s[i] == 'U' &&
                        dp[i][j][k][0] < dp[i - 1][0][k][0] + 1) {
                        dp[i][j][k] = {dp[i - 1][0][k][0] + 1, i - 1, 0, k};
                    }
                } else {
                    // P
                    dp[i][j][k][0]--;
                    if (s[i] == 'P' &&
                        dp[i][j][k][0] < dp[i - 1][1][k][0] + 1) {
                        dp[i][j][k] = {dp[i - 1][1][k][0] + 1, i - 1, 1, k};
                    }
                }
            }
        }
    }

    if (dp[N - 1][2][99][0] > 0) {
        cout << dp[N - 1][2][99][0] - 100 << "\n";
        vector<vector<int>> resv(100, vector<int>(3));
        int i = N - 1, j = 2, k = 99;
        while (i >= 0) {
            // cout << i << " " << j << " " << k << endl;
            int ni = dp[i][j][k][1];
            int nj = dp[i][j][k][2];
            int nk = dp[i][j][k][3];

            if (j != nj || k != nk) {
                resv[k][j] = i + 1;
            }

            i = ni, j = nj, k = nk;
        }

        for (auto t : resv) {
            for (auto u : t) {
                cout << u << " ";
            }
            cout << "\n";
        }
    } else {
        cout << -1;
    }
    return 0;
}
